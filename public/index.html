1 Agent calls your API

3 Agent signs USDC payment off-chain

6 Your API serves the response. You get paid.

Non-Custodial

Integrate in minutes.

Three ways to add PayGate to your stack.

// Merchant: verify + settle an agent's payment
const res = await fetch('https://paygate.arispay.app/facilitator/settle', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({
    x402Version: 2,
    paymentPayload,     // from the agent's X-PAYMENT header
    paymentRequirements // your 402 response requirements
  })
});

const { success, transaction } = await res.json();
// success: true, transaction: "0xabc..."
# Register your API with PayGate
curl -X POST https://paygate.arispay.app/api/merchants \
  -H "Content-Type: application/json" \
  -d '{"name":"My API","email":"dev@example.com","wallet":"0x..."}'

# Add a paywalled endpoint ($0.01 per call)
curl -X POST https://paygate.arispay.app/api/endpoints \
  -H "Authorization: Bearer agfac_..." \
  -d '{"path":"/v1/data","targetUrl":"https://api.example.com/v1/data",
       "price":"10000","description":"Get data","method":"GET"}'

# Agents now call: https://paygate.arispay.app/my-api/v1/data
# First call → 402. Pay → proxied to your real API.
// Agent: pay for an API call with payagent
import { payFetch } from 'payagent';

const response = await payFetch('https://api.example.com/v1/data', {
  wallet: myWallet,
  maxBudget: '1.00', // max $1.00 USDC
});

console.log(response.data);
// Payment handled automatically — 402 → sign → retry
